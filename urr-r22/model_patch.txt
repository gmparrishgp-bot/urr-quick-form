 async function ensureHandwritingModel(){
   if(ocr)return;
   if(aiLoading)return aiLoading;
   aiLoading=(async()=>{
     setAIProgress(3,"Loading form OCR…");
     paddleModule=await import("https://cdn.jsdelivr.net/npm/@paddleocr/paddleocr-js/+esm");
     const {PaddleOCR}=paddleModule;
     if(!PaddleOCR)throw new Error("PaddleOCR browser SDK did not load.");
     ocr=await PaddleOCR.create({
       textDetectionModelName:"PP-OCRv5_mobile_det",
       textRecognitionModelName:"PP-OCRv5_mobile_rec",
       textDetectionBatchSize:1,
       textRecognitionBatchSize:8,
       ortOptions:{backend:"wasm",wasmPaths:"https://cdn.jsdelivr.net/npm/onnxruntime-web/dist/",numThreads:1,simd:true}
     });
     setAIProgress(12,"Form OCR ready.","ready");
   })().catch(e=>{aiLoading=null;throw e;});
   return aiLoading;
 }
 async function ensureTrOCR(){
   if(trocrPipe)return trocrPipe;
   if(trocrLoading)return trocrLoading;
   trocrLoading=(async()=>{
     setAIProgress(13,"Loading handwritten-line reader…");
     const mod=await import("https://cdn.jsdelivr.net/npm/@xenova/transformers@2.17.2/+esm");
     if(!mod?.pipeline)throw new Error("Handwritten-line reader did not load.");
     if(mod.env){mod.env.allowLocalModels=false;mod.env.useBrowserCache=true;}
     trocrPipe=await mod.pipeline("image-to-text","Xenova/trocr-small-handwritten",{
       quantized:true,
       progress_callback:p=>{if(p&&typeof p.progress==="number")setAIProgress(13+Math.min(9,p.progress*.09),`Loading handwritten-line reader… ${Math.round(p.progress)}%`);}
     });
     setAIProgress(22,"Handwritten-line reader ready.","ready");
     return trocrPipe;
   })().catch(e=>{trocrLoading=null;throw e;});
   return trocrLoading;
 }
 async function trocrReadLines(metas,label,p0,p1){
   await ensureTrOCR();
   const out=[];
   for(let i=0;i<metas.length;i++){
     const m=metas[i];setAIProgress(p0+(p1-p0)*(i/Math.max(1,metas.length)),`${label} ${i+1} of ${metas.length}…`);
     try{
       const data=m.canvas.toDataURL("image/png"),r=await withTimeout(trocrPipe(data,{max_new_tokens:56}),45000,`${label} ${i+1}`);
       const text=String(Array.isArray(r)?r[0]?.generated_text:r?.generated_text||"").replace(/\s+/g," ").trim();
       out.push({meta:m,text});
     }catch(e){out.push({meta:m,text:"",error:String(e?.message||e)});}
     await new Promise(r=>setTimeout(r,0));
   }
   return out;
 }
