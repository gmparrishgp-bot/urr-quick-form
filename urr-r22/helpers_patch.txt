 function makePaddleVariant(canvas,x0,y0,x1,y1,kind="contrast",targetW=1500){
   const sx=Math.max(0,Math.floor(x0*canvas.width)),sy=Math.max(0,Math.floor(y0*canvas.height));
   const sw=Math.max(1,Math.floor((x1-x0)*canvas.width)),sh=Math.max(1,Math.floor((y1-y0)*canvas.height));
   const scale=Math.min(4,targetW/sw);
   const c=document.createElement("canvas");c.width=Math.max(180,Math.round(sw*scale));c.height=Math.max(70,Math.round(sh*scale));
   const x=c.getContext("2d",{willReadFrequently:true,alpha:false});x.fillStyle="#fff";x.fillRect(0,0,c.width,c.height);x.imageSmoothingEnabled=true;x.imageSmoothingQuality="high";x.drawImage(canvas,sx,sy,sw,sh,0,0,c.width,c.height);
   if(kind!=="original"){
     const im=x.getImageData(0,0,c.width,c.height),d=im.data;
     for(let k=0;k<d.length;k+=4){const r=d[k],g=d[k+1],b=d[k+2];let y=.299*r+.587*g+.114*b;
       if(kind==="contrast")y=Math.max(0,Math.min(255,(y-128)*1.65+128));
       else if(kind==="threshold")y=y<185?0:255;
       else if(kind==="soft")y=Math.max(0,Math.min(255,(y-70)*1.3));
       else if(kind==="dark"){
         const spread=Math.max(r,g,b)-Math.min(r,g,b),redLike=r>70&&r>g*1.07&&r>b*1.03&&(r-g)>8,purpleLike=b>r*1.04&&b>g*1.02&&(b-r)>6;
         const keep=y<235&&!redLike&&!purpleLike&&(spread<70||y<150);y=keep?Math.max(0,Math.min(255,(y-105)*1.75+105)):255;
       }
       d[k]=d[k+1]=d[k+2]=y;d[k+3]=255;
     }x.putImageData(im,0,0);
   }
   return{canvas:c,x0,y0,x1,y1,kind};
 }

 function normalizeQuoteText(s){
   s=String(s||"").replace(/[Oo]/g,"0").replace(/#/g,"$").replace(/\bH(?:E|C)\b/ig,"HR").replace(/,/g,".");
   s=s.replace(/\$[Il](\d)/g,(_,d)=>"$1"+d).replace(/\$E(\d)/gi,(_,d)=>"$6"+d).replace(/\bE(\d)\s*H(?:R|RC|RCE|E|C)\b/gi,".$1 HR").replace(/\bR?E(\d)\s*H(?:R|RC|RCE|E|C)\b/gi,".$1 HR");
   return s.replace(/\s+/g," ").trim();
 }
 function quoteCandidatesFromText(s){

 }
