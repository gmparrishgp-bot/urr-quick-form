 async function analyzePage(kind,img,paper){
   setAIProgress(2,"Preparing URR image…");
   const canvas=normalizePaperCanvas(img,paper),cols=detectHeaderColumns(canvas);await ensureHandwritingModel();
   let rowGap=detectGridGap(canvas);if(!(rowGap>.022&&rowGap<.050))rowGap=.0355;const geometry=buildPhysicalRows(canvas,rowGap),rows=geometry.rows;
   const roMeta=makePageCrop(canvas,cols.ro.x0,.112,cols.ro.x1,.225,700);
   const roArr=await withTimeout(ocr.predict(roMeta.canvas,{textDetLimitSideLen:900,textDetLimitType:"max",textRecScoreThresh:.08}),45000,"RO read");
   let ro="";if(roArr?.[0]?.items){const items=remapCropItems(roArr[0],roMeta),txt=orderedItemsText(items);const m=txt.match(/\bRO\s*#?\s*[:#-]?\s*(\d{4,8})\b/i);if(m)ro=m[1];if(!ro){const nums=items.map(x=>String(x.text||"").replace(/\D/g,"")).filter(x=>/^\d{4,6}$/.test(x));if(nums.length===1)ro=nums[0];}}
   const descStart=cols.make.x0+.018,partX0=cols.ro.x1-.145,totalX0=cols.date.x0,totalX1=cols.date.x1,fixX0=cols.tech.x0,fixX1=(cols.tech.x0+cols.tech.x1)/2,asisX0=fixX1,asisX1=cols.tech.x1;
   const metas=[];for(let i=0;i<rows.length;i++){for(const kindv of ["contrast","threshold"]){const m=makePaddleVariant(canvas,descStart,rows[i].top,totalX1,rows[i].bottom,kindv,1700);m.row=i;metas.push(m);}}
   const reads=await paddleBatch(metas,"Reading handwritten URR rows",22,88),byRow=Array.from({length:rows.length},()=>[]);for(const r of reads)if(r)byRow[r.meta.row].push(r);
   const rowData=[];
   for(let i=0;i<rows.length;i++){
     const variants=byRow[i],descTexts=[],quoteTexts=[],totalTexts=[];
     for(const v of variants){descTexts.push(orderedItemsText(v.items.filter(x=>x.nx<partX0-.004)));quoteTexts.push(orderedItemsText(v.items.filter(x=>x.nx>=partX0-.025&&x.nx<totalX0-.006)));totalTexts.push(orderedItemsText(v.items.filter(x=>x.nx>=totalX0-.015)));}
     const desc=bestDescription(descTexts);let totalCands=totalTexts.flatMap(totalCandidatesFromText),quote=chooseQuote(quoteTexts,totalCands),totalInfo=chooseTotal(totalTexts,quote);
     if(totalInfo.kind==="PRICE"){if(!quote.hours&&quote.parts){const h=inferHoursFromTotal(totalInfo.value,quote.parts);if(h)quote.hours=h;}else if(quote.hours&&!quote.parts){const p=inferPartsFromTotal(totalInfo.value,quote.hours);if(p)quote.parts=p;}}
     const red=redInkDensity(canvas,partX0-.025,rows[i].top,totalX0-.006,rows[i].bottom),descInk=descriptionInkDensity(canvas,rows[i].y,rowGap,descStart,partX0-.01);let fixInk=0,asisInk=0;if(kind==="approval"){fixInk=cellInkDensity(canvas,fixX0,rows[i].top,fixX1,rows[i].bottom);asisInk=cellInkDensity(canvas,asisX0,rows[i].top,asisX1,rows[i].bottom);}
     const hasDesc=desc.score>.65||descInk>.0017,approvalEvidence=kind==="approval"&&(fixInk>.12||asisInk>.12),endEvidence=!!quote.parts||!!quote.hours||!!totalInfo.kind||red>.0025||approvalEvidence;
     rowData.push({title:desc.text,titleScore:desc.score,parts:quote.parts||"",hours:quote.hours||"",totalInfo,red,descInk,fixInk,asisInk,hasDesc,endEvidence});
   }
   const jobs=[];let pending=null;
   function finish(force=false){if(!pending)return;const title=cleanText(pending.pieces.join(" ")),titleLow=force||!title||pending.score<2.3,issues=[];if(titleLow)issues.push("description");if(pending.hoursLow)issues.push("labor");if(pending.partsLow)issues.push("parts");jobs.push({num:jobs.length+1,status:pending.status||"PENDING",title:titleLow?"[HANDWRITING NEEDS REVIEW]":title,hours:pending.hours||"",parts:pending.parts||"",confidence:titleLow?.42:.88,titleLow,hoursLow:!!pending.hoursLow,partsLow:!!pending.partsLow,comment:[issues.length?`Verify ${[...new Set(issues)].join(", ")} against photo.`:"",...(pending.notes||[])].filter(Boolean).join(" ")});pending=null;}
   for(const r of rowData){if(!r.hasDesc&&!r.endEvidence){if(pending)finish(true);continue;}if(!pending)pending={pieces:[],score:0,hours:"",parts:"",status:"PENDING",hoursLow:false,partsLow:false,notes:[]};if(r.hasDesc&&r.title){pending.pieces.push(r.title);pending.score+=Math.max(0,r.titleScore);}if(r.parts)pending.parts=r.parts;if(r.hours)pending.hours=r.hours;if(r.totalInfo.kind==="RECON"||/\brecon\b/i.test(r.title||""))pending.status="RECON";else if(r.totalInfo.kind==="NOTE")pending.status="NOTE";if(r.totalInfo.kind==="PRICE"&&pending.hours!==""&&pending.parts!==""&&calc(pending.hours,pending.parts)!==r.totalInfo.value){pending.hoursLow=true;pending.partsLow=true;pending.notes.push(`Calculated ${money(calc(pending.hours,pending.parts))} does not match sheet total ${money(r.totalInfo.value)}.`);}if(kind==="approval"){if(r.fixInk>r.asisInk*1.35&&r.fixInk>.12)pending.status="APPROVED";else if(r.asisInk>r.fixInk*1.35&&r.asisInk>.12)pending.status="DECLINED";}if(r.endEvidence)finish(false);}if(pending)finish(true);
   const below=detectBelowRed(canvas,geometry.tableBottom+.004);for(const sig of below){const y=sig.y;if(y>.985||descriptionInkDensity(canvas,y,rowGap,.035,.72)<.010)continue;const top=Math.max(geometry.tableBottom,y-rowGap*.58),bottom=Math.min(.992,y+rowGap*.52),ms=["contrast","threshold"].map(k=>makePaddleVariant(canvas,.035,top,.80,bottom,k,1800));const rr=await paddleBatch(ms,"Reading repair below form",89,96),texts=rr.map(x=>orderedItemsText(x.items)),desc=bestDescription(texts),tc=texts.flatMap(totalCandidatesFromText),q=chooseQuote(texts,tc),ti=chooseTotal(texts,q),low=desc.score<2;jobs.push({num:jobs.length+1,status:ti.kind==="RECON"?"RECON":ti.kind==="NOTE"?"NOTE":"PENDING",title:low?"[HANDWRITING NEEDS REVIEW]":desc.text,hours:q.hours||"",parts:q.parts||"",confidence:low?.42:.86,titleLow:low,hoursLow:false,partsLow:false,comment:low?"Verify description against photo.":""});}
   if(!jobs.length)throw new Error("No repair jobs were recognized.");
   return{ro,jobs,notes:[`${jobs.length} repair${jobs.length===1?"":"s"} reconstructed from the actual form rows.`,`PaddleOCR contrast + threshold reads are combined with RV context and the written Total Cost as a cross-check.`,`RO is restricted to the fixed RO# header column.`]};
 }
