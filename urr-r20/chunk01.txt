 function detectPaperRect(img){
   const max=600,s=Math.min(1,max/Math.max(img.naturalWidth,img.naturalHeight));
   const c=document.createElement("canvas");
   c.width=Math.max(1,Math.round(img.naturalWidth*s));c.height=Math.max(1,Math.round(img.naturalHeight*s));
   const ctx=c.getContext("2d",{willReadFrequently:true});ctx.drawImage(img,0,0,c.width,c.height);
   const d=ctx.getImageData(0,0,c.width,c.height).data,xs=[],ys=[];
   for(let y=0;y<c.height;y+=2)for(let x=0;x<c.width;x+=2){
     const i=(y*c.width+x)*4,r=d[i],g=d[i+1],b=d[i+2],lum=(r+g+b)/3,spread=Math.max(r,g,b)-Math.min(r,g,b);
     if(lum>138&&spread<90){xs.push(x);ys.push(y);}
   }
   if(xs.length<500)return{x:0,y:0,w:img.naturalWidth,h:img.naturalHeight};
   xs.sort((a,b)=>a-b);ys.sort((a,b)=>a-b);const q=(a,p)=>a[Math.max(0,Math.min(a.length-1,Math.floor(a.length*p)))];
   let x0=q(xs,.012)/s,x1=q(xs,.988)/s,y0=q(ys,.008)/s,y1=q(ys,.994)/s;
   if(x1-x0<img.naturalWidth*.48||y1-y0<img.naturalHeight*.58)return{x:0,y:0,w:img.naturalWidth,h:img.naturalHeight};
   const px=(x1-x0)*.012,py=(y1-y0)*.01;
   return{x:Math.max(0,x0-px),y:Math.max(0,y0-py),w:Math.min(img.naturalWidth-x0+px,x1-x0+2*px),h:Math.min(img.naturalHeight-y0+py,y1-y0+2*py)};
 }

 function polyBox(item,w,h){
   const p=item?.poly||item?.polygon||[];let pts=[];
   if(Array.isArray(p)){
     for(const q of p){
       if(Array.isArray(q)&&q.length>=2)pts.push([Number(q[0]),Number(q[1])]);
       else if(q&&typeof q==="object")pts.push([Number(q.x??q[0]),Number(q.y??q[1])]);
     }
     if(!pts.length&&p.length>=8&&typeof p[0]==="number")for(let i=0;i+1<p.length;i+=2)pts.push([Number(p[i]),Number(p[i+1])]);
   }
   if(!pts.length)return{x0:0,y0:0,x1:w,y1:h,cx:w/2,cy:h/2,bw:w,bh:h};
   const xs=pts.map(x=>x[0]),ys=pts.map(x=>x[1]);const x0=Math.min(...xs),x1=Math.max(...xs),y0=Math.min(...ys),y1=Math.max(...ys);
   return{x0,y0,x1,y1,cx:(x0+x1)/2,cy:(y0+y1)/2,bw:x1-x0,bh:y1-y0};
 }
