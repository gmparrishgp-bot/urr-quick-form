     }
   });

     if(!d)return;

 function apply(kind,out){
   ensureRO(out);

   if(kind==="quote"){
     const incoming=Array.isArray(out?.jobs)?out.jobs:[];
     if(!incoming.length)throw new Error("No repair jobs were read from the quote photo.");
     const startNum=Array.isArray(state.jobs)?state.jobs.length:0;
     const mapped=incoming.map((j,i)=>({
       num:startNum+i+1,
       // Quote dollars/labor do NOT make something RECON. Only an explicit read does.
       status:["RECON","NOTE"].includes(String(j?.status||"").toUpperCase())?String(j.status).toUpperCase():"PENDING",
       title:String(j?.title||""),
       hours:j?.hours==null?"":String(j.hours),
       parts:j?.parts==null?"":String(j.parts),
       confidence:Number.isFinite(Number(j?.confidence))?Number(j.confidence):0.55,
       titleLow:!!j?.titleLow,hoursLow:!!j?.hoursLow,partsLow:!!j?.partsLow,
       comment:String(j?.comment||"")
     }));
     state.jobs=[...(Array.isArray(state.jobs)?state.jobs:[]),...mapped];
     state.sheetCount=(Number(state.sheetCount)||0)+1;
     state.notes=[...(state.notes||[]),`Quote sheet ${state.sheetCount}: ${(out?.notes||[]).join(" ")}`];
   }else{
     const incoming=Array.isArray(out?.jobs)?out.jobs:[];
     if(!incoming.length)throw new Error("No repair jobs were read from the approval photo.");

     if(!Array.isArray(state.jobs)||!state.jobs.length){
       // Approval-only workflow: the approval sheet becomes the source document.
       // This is valid when the original quote was never scanned.
       state.jobs=incoming.map((j,i)=>({
         num:Number(j?.num)||i+1,
         status:["APPROVED","DECLINED","RECON","NOTE","PENDING"].includes(String(j?.status||"").toUpperCase())
           ?String(j.status).toUpperCase()
           :"PENDING",
         title:String(j?.title||""),
         hours:j?.hours==null?"":String(j.hours),
         parts:j?.parts==null?"":String(j.parts),
         confidence:Number.isFinite(Number(j?.confidence))?Number(j.confidence):0.55,
         titleLow:!!j?.titleLow,
         hoursLow:!!j?.hoursLow,
         partsLow:!!j?.partsLow,
         comment:String(j?.comment||"")
       }));
       state.sheetCount=1;
       state.notes=[
         "Approval-only workflow: no quote photo was loaded; jobs were created directly from the approval sheet.",
         ...(Array.isArray(out?.notes)?out.notes.map(String):[])
       ];
     }else{
       reconcileApproval(incoming);
       state.notes=[...(state.notes||[]),...(Array.isArray(out?.notes)?out.notes.map(String):[])];
     }
   }

   render();
 }

 async function analyzePhoto(kind,imageData){
   currentPhoto={kind,imageData};
   sourceWrap.style.display="block";
   const thumb=document.createElement("img");
   thumb.src=imageData;
   thumb.alt=kind==="approval"?"Approval sheet":"URR sheet";
   sourceGallery.appendChild(thumb);
   try{
     const img=await loadImg(imageData),paper=detectPaperRect(img);
     const result=await analyzePage(kind,img,paper);
     const count=result.jobs.length;
     apply(kind,result);
     const rendered=document.querySelectorAll("#jobs .job").length;
     setAIProgress(100,
       kind==="quote"
         ?`Quote sheet ${state.sheetCount}: ${count} repair${count===1?"":"s"} added. ${rendered} total loaded.`
         :`${state.sheetCount===1&&state.notes?.[0]?.startsWith("Approval-only workflow") ? "Approval sheet loaded" : "Approval reconciled"}. ${rendered} repair${rendered===1?"":"s"} loaded.`,
       "ready"
     );
     if(conn&&conn.open)conn.send({type:"analysis_done",ro:state.ro||""});
     ds.textContent="Updated from phone";ds.className="status sync";
     setTimeout(()=>{if(conn&&conn.open){ds.textContent="Phone connected";ds.className="status connected";}},1200);
   }catch(e){
     console.error(e);
     setAIProgress(parseInt(aiProgress.style.width)||5,"URR read stopped: "+(e?.message||String(e)),"error");
     if(conn&&conn.open)conn.send({type:"analysis_error",message:e?.message||String(e)});
   }
 }
