function normKey(t){return String(t||"").toUpperCase().replace(/[^A-Z0-9]+/g," ").trim();}

function titleSimilarity(a,b){

 let same=0;for(const x of A)if(B.has(x))same++;
 return same/Math.max(A.size,B.size);
}

function normalized(t){return String(t||"").replace(/curb\s*side|door\s*side/gi,"DS").replace(/driver\s*side|off\s*door\s*side/gi,"ODS").replace(/wheel\s*bearing\s*pack/gi,"WBP").replace(/\s+/g," ").trim();}
function findRO(t){for(const p of [/\bRO\s*(?:#|NO\.?|NUMBER)?\s*[:#-]?\s*(\d{4,8})\b/i,/\bREPAIR\s+ORDER\s*[:#-]?\s*(\d{4,8})\b/i,/\bWO\s*[:#-]?\s*(\d{4,8})\b/i]){const m=t.match(p);if(m)return m[1]}return "";}
function hours(line){for(const p of [/\b(?:LABOR|LAB|HRS?|HOURS?)\s*[:=-]?\s*(\.?\d+(?:\.\d+)?)\b/i,/\b(\d+(?:\.\d+)?)\s*(?:HRS?|HOURS?)\b/i]){const m=line.match(p);if(m)return m[1].replace(/^\./,"0.")}return "";}
function parts(line){for(const p of [/\b(?:PARTS?|PTS?)\s*[:=-]?\s*\$?\s*(\d+(?:\.\d{1,2})?)/i,/\$\s*(\d+(?:\.\d{1,2})?)/]){const m=line.match(p);if(m)return m[1]}return "";}
function strip(line){return normalized(line.replace(/\b(?:LABOR|LAB|HRS?|HOURS?|PARTS?|PTS?)\b\s*[:=-]?\s*\$?\s*\.?\d+(?:\.\d+)?/gi," ").replace(/\$\s*\d+(?:\.\d+)?/g," ").replace(/\b\d+(?:\.\d+)?\s*(?:HRS?|HOURS?)\b/gi," "));}
function parseQuote(text){
 const lines=text.replace(/\r/g,"").split("\n").map(x=>x.trim()).filter(Boolean),jobs=[];let pending="";
 for(const raw of lines){let line=normalized(raw);if(line.length<3||/^(RO|REPAIR ORDER|CUSTOMER|VIN|STOCK|DATE|TECH|TECHNICIAN)\b/i.test(line))continue;
 const nm=line.match(/^\s*(\d{1,2})[.)-]?\s+(.+)/);let num="",body=line;if(nm){num=nm[1];body=nm[2]}
 const h=hours(line),p=parts(line),meaning=/[A-Za-z]{3,}/.test(body);
 if((h||p)&&meaning){let title=strip(body).replace(/^[-:]+|[-:]+$/g,"").trim();if(pending){title=(pending+" "+title).trim();pending=""}jobs.push({num:num||jobs.length+1,title:title||"REVIEW TECH DESCRIPTION",hours:h,parts:p,status:"PENDING",confidence:.7,comment:"Browser-read result — verify handwriting."});}
 else if(num&&meaning){pending=body;} else if(meaning&&line.length>12){pending=pending?pending+" "+line:line;}
 }
 if(!jobs.length){lines.filter(x=>/[A-Za-z]{4,}/.test(x)).slice(0,12).forEach((x,i)=>jobs.push({num:i+1,title:strip(x),hours:hours(x),parts:parts(x),status:"PENDING",confidence:.45,comment:"Low-confidence browser read."}));}
 return {ro:findRO(text),jobs,notes:["Local browser OCR used. Verify flagged fields before DMS copy."]};
}
function parseApproval(text){
 const lines=text.replace(/\r/g,"").split("\n").map(normalized).filter(Boolean),jobs=[];let last="";
 for(const line of lines){const nm=line.match(/^\s*(\d{1,2})[.)-]?\s+/);if(nm)last=nm[1];let s="";if(/\b(?:FIX|APPROVE[DS]?|APPROVED|YES)\b/i.test(line))s="APPROVED";else if(/\b(?:AS\s*IS|DECLINE[DS]?|DECLINED|NO)\b/i.test(line))s="DECLINED";else if(/\bRECON\b/i.test(line))s="RECON";else if(/\bNOTE\b/i.test(line))s="NOTE";if(s)jobs.push({num:last||jobs.length+1,status:s,title:strip(line)});}
 return {ro:findRO(text),jobs,notes:jobs.length?[]:["Approval marks were not read confidently. Review statuses."]};
}
