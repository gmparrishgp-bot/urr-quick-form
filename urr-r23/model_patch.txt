 async function ensureHandwritingModel(){
   if(ocr&&trocr)return;
   if(aiLoading)return aiLoading;
   aiLoading=(async()=>{
     setAIProgress(3,"Loading form reader…");
     paddleModule=await import("https://cdn.jsdelivr.net/npm/@paddleocr/paddleocr-js/+esm");
     const {PaddleOCR}=paddleModule;
     if(!PaddleOCR)throw new Error("PaddleOCR browser SDK did not load.");
     ocr=await PaddleOCR.create({
       textDetectionModelName:"PP-OCRv5_mobile_det",
       textRecognitionModelName:"PP-OCRv5_mobile_rec",
       textDetectionBatchSize:1,
       textRecognitionBatchSize:8,
       ortOptions:{backend:"wasm",wasmPaths:"https://cdn.jsdelivr.net/npm/onnxruntime-web/dist/",numThreads:1,simd:true}
     });
     setAIProgress(22,"Form reader ready. Loading handwriting reader…");
     hfModule=await import("https://cdn.jsdelivr.net/npm/@huggingface/transformers@3.8.1");
     try{
       if(hfModule.env?.backends?.onnx?.wasm){
         hfModule.env.backends.onnx.wasm.numThreads=1;
         hfModule.env.backends.onnx.wasm.simd=true;
         hfModule.env.backends.onnx.wasm.wasmPaths="https://cdn.jsdelivr.net/npm/onnxruntime-web/dist/";
       }
       if(hfModule.env)hfModule.env.allowLocalModels=false;
     }catch{}
     const {pipeline}=hfModule;
     trocr=await pipeline("image-to-text","Xenova/trocr-small-handwritten",{
       device:"wasm",
       dtype:"q8",
       progress_callback:x=>{
         if(x&&x.status==="progress"&&Number.isFinite(x.progress))setAIProgress(22+Math.round(x.progress*.28),`Loading handwriting reader… ${Math.round(x.progress)}%`);
       }
     });
     setAIProgress(52,"Handwriting reader ready.","ready");
   })().catch(e=>{aiLoading=null;throw e;});
   return aiLoading;
 }
